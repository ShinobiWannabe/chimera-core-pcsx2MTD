option('minibox_dir', type: 'string', value: '', description: 'miniBox checkout (guest kit + host)')

option('mesa_guest_dir', type: 'string', value: '',
  description: 'A Mesa built for the guest (see ~/minihawk-tools/mesa-guest). With one, this core also builds PCSX2\'s OpenGL renderer and runs it against Mesa softpipe INSIDE the sandbox - a real GL implementation, no GPU, still deterministic. Without one, only the software renderer is built.')

option('gl_bridge', type: 'boolean', value: false,
  description: 'EXPERIMENT: let the sandboxed renderer drive a REAL GPU on the host, for speed. The GPU is outside the sandbox, outside the savestate and different on every machine, so a movie recorded this way carries no promise that it replays. Builds the guest wrappers into the core and the host half into run-wbx; the softpipe is still there and still the default.')

option('jit_rasterizer', type: 'boolean', value: true,
  description: 'Draw the software renderer\'s scanlines with PCSX2\'s own code generator (the default, in the sandbox too) rather than its C++ fallback. Both draw the same bytes; the generated code is faster. The code generator writes into the code region the recompilers already use, and the processor it targets is pinned (waterbox/host-sys.cpp), so every computer generates the same code. false builds the C++ path, which run-gate.sh\'s rasteriser leg compares against.')

option('raster_bands', type: 'integer', min: 1, max: 64, value: 8,
  description: 'How many bands the software rasteriser draws in, each on its own parallel task. Never the host\'s worker count: the picture is the same at any value, but the bands live in the machine\'s memory, so a different value is a different core identity (savestates do not cross). More bands only help a host with more cores than bands.')
option('raster_band_height', type: 'integer', min: 0, max: 8, value: 0,
  description: 'log2 of the scanlines in each band\'s strip; 0 keeps PCSX2\'s own (16 lines).')
